cmake_minimum_required(VERSION 3.16)
project(plugin_gc CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
add_executable(harness host/harness.cpp)
add_library(control SHARED host/control.cpp)
find_package(Threads REQUIRED)
target_link_libraries(control Threads::Threads)
if(WIN32)
  target_link_libraries(harness avrt winmm)
else()
  target_link_libraries(harness ${CMAKE_DL_LIBS} Threads::Threads)
endif()
if(MSVC)
  target_compile_options(harness PRIVATE /O2)
  target_compile_options(control PRIVATE /O2 /fp:fast)
else()
  target_compile_options(control PRIVATE -O2)
endif()
