# Neo Analyzer factory presets. Rebuild: python3 tools/gen-machine-presets.py Machines/NeoAnalyzer
# The audio is never touched; presets set how the analyzer shows it and the loudness target.
machine "Neo Analyzer"

# Mixing: headroom for mastering (about −18 LUFS), a medium spectrum tilted so a finished balance reads flat.
preset "Mix"
Smoothing = 1/6 oct
Speed = Medium
Peak hold = 2
Tilt = 3 dB/oct
Target = -18

# Mastering for streaming: −14 LUFS, a slow, fine spectrum leaning bright to judge the top end.
preset "Mastering"

Peak hold = 10
Tilt = 4.5 dB/oct
Target = -14

Tilt = 0 dB/oct
